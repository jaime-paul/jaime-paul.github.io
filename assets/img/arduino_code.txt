// Motor control pins into the arduino
#define STEP_PIN 4
#define DIR_PIN 3
 
 
// Input pins into the arduino
#define LATCHING_BUTTON_PIN 5
#define LIMIT_SWITCH_PIN 6
 
 
// LED pins - input pins into the arduino
#define LED_RED_PIN 8
#define LED_GREEN_PIN 9
#define LED_BLUE_PIN 10
 
//extra credit input pins into the arduino (non-latching button inputs)
#define CCW_PIN 11 //clockwise and counterclockwise
#define CW_PIN 12
 
 
float largeSyringeDiameter = 18.7; //measured internal diameter of syringe
float smallSyringeDiameter = 14.6; 
 
 
float flowRate = 5; //mL/min [EDIT THIS VALUE to change pump speed]
float syringeDiameter = smallSyringeDiameter; //(smallSyringeDiameter for 10 mL, largeSyringeDiameter for 20 mL)
float distancePerRevolution = 8.0; //lead screw lead
float stepsPerRevolution = 3200; //3200 (16 steps) microstepping 
 
 
#include <AccelStepper.h>
 
 
AccelStepper stepper(AccelStepper::DRIVER, 4 , 3); //used the STEP and DIR PIN input values
 
void setup() //initial set up: configuring pins to input_pullup/output 
{
stepper.setMaxSpeed(1000); //loop only runs once, so this just initializing values
calculateStepperSpeed();
pinMode(LED_RED_PIN, OUTPUT);
pinMode(LED_GREEN_PIN, OUTPUT); //LED’s outputs, buttons are INPUT_pullup
pinMode(LED_BLUE_PIN, OUTPUT);
pinMode(LATCHING_BUTTON_PIN, INPUT_PULLUP);
pinMode(LIMIT_SWITCH_PIN, INPUT_PULLUP);
setColor(0, 255, 0); //green initially 
pinMode(CCW_PIN, INPUT_PULLUP);
pinMode(CW_PIN,INPUT_PULLUP);
}
 
void loop() //runs infinitely, constantly checking for changes in button states 
{
 
if (digitalRead(LATCHING_BUTTON_PIN) == LOW && digitalRead(LIMIT_SWITCH_PIN)==HIGH){ //if latching button is pressed, run the pump
  calculateStepperSpeed();
  stepper.runSpeed();
  setColor(0, 255, 0); // Green Color
}
if ((digitalRead(LATCHING_BUTTON_PIN) == HIGH && digitalRead(LIMIT_SWITCH_PIN)== HIGH)){ //if latching button is pressed again, pause the pump
  setColor(255, 50, 0); // Yellow Color
  if (digitalRead(CCW_PIN) == LOW){ //if its paused and the non-latching (1) is pressed
   setColor(255, 0, 255); //purple CCW //reverse for easy reset
   stepper.setSpeed(900);
   stepper.runSpeed();
  }
  if (digitalRead(CW_PIN) == LOW ){ //same thing for the other non-latching (2)
   setColor(0, 0, 255); //blue CW //forward if reversed too far
   stepper.setSpeed(-900); // neg bc this is the forward direction
   stepper.runSpeed();
  }
}
if (digitalRead(LATCHING_BUTTON_PIN) == LOW && digitalRead(LIMIT_SWITCH_PIN)==LOW){ //If the limit switch is triggered, stop the pump, it’s empty
  setColor(255, 0, 0); // red Color
}
if (digitalRead(LATCHING_BUTTON_PIN) == HIGH && digitalRead(CCW_PIN) == LOW){ //pause after limit switch red, then can reverse
 if (digitalRead(CCW_PIN) == LOW){
   stepper.setSpeed(900);
   stepper.runSpeed();
  }
 
}
 
}
void setColor(int redValue, int greenValue, int blueValue) {
analogWrite(LED_RED_PIN, redValue); //simple set color function to make setting RGB values easier in void loop
analogWrite(LED_GREEN_PIN, greenValue);
analogWrite(LED_BLUE_PIN, blueValue);
}
 
void calculateStepperSpeed() { //calculating stepper speed (see calculation section for explanation)
float syringeArea = 3.14159 * pow(syringeDiameter / 2.0, 2);
float flowRateMmCubed = flowRate * 1000.0;
float linearSpeed = flowRateMmCubed / syringeArea;
float stepperSpeed = (linearSpeed / distancePerRevolution) * stepsPerRevolution / 60.0; //steps per second
stepper.setSpeed(-stepperSpeed); //neg bc this is the forward direction
 
}
